Catatan penting

Ini prototipe. Template biometrik disimpan di localStorage pada browser/perangkat ini saja, bukan server dan tidak tersinkron antarperangkat. Hasil pencocokan dapat salah; guru wajib memeriksa identitas. Gunakan data uji, siapkan persetujuan yang sesuai, pembatasan akses, penghapusan dan retensi data, serta absensi manual sebelum dipakai di sekolah. Jangan simpan biometrik siswa nyata pada prototipe ini.

'+esc(s.cls)+''+(session[s.id]?'Hadir '+esc(session[s.id].time):'Belum hadir')+'').join('')||'Belum ada siswa.'} $('reset').onclick=()=>{if(confirm('Reset status hadir sesi ini?')){session={};render()}};$('export').onclick=()=>{let data=[['Nama','Kode','Kelas','Tanggal','Waktu','Mapel','Status'],...students.map(s=>[s.name,s.code,s.cls,session[s.id]?.date||'',session[s.id]?.time||'',session[s.id]?.subject||$('subject').value,session[s.id]?'Hadir':'Belum hadir'])];let csv=data.map(r=>r.map(x=>'"'+String(x??'').replace(/"/g,'""')+'"').join(',')).join('\\r\\n'),a=document.createElement('a');a.href=URL.createObjectURL(new Blob(['\\ufeff'+csv],{type:'text/csv;charset=utf-8'}));a.download='rekap-absensi-wajah.csv';a.click();URL.revokeObjectURL(a.href)};window.addEventListener('beforeunload',()=>stream?.getTracks().forEach(t=>t.stop()));render();init();
